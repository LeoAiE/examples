import json
import os
from typing import Dict, Any, List
from openai import OpenAI

class BaseAgent:
    def __init__(self, name: str, db_path: str, role_description: str):
        """
        Initialize base agent
        
        Args:
            name: Agent name (e.g., "Travel Agent")
            db_path: Path to the agent's database file
            role_description: Description of what this agent does
        """
        self.name = name
        self.db_path = db_path
        self.role_description = role_description
        self.client = None
        self.model = "gpt-4o-mini"
        self.database_content = self.load_database()
    
    def initialize_openai(self, api_key: str, model: str = "gpt-4o-mini"):
        """Initialize OpenAI client"""
        self.client = OpenAI(api_key=api_key)
        self.model = model
    
    def load_database(self) -> str:
        """Load the database content from text file"""
        try:
            if os.path.exists(self.db_path):
                with open(self.db_path, 'r', encoding='utf-8') as file:
                    content = file.read()
                    return content if content else "Database file is empty"
            else:
                # Create empty file if doesn't exist
                os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
                with open(self.db_path, 'w', encoding='utf-8') as file:
                    file.write("")
                return "Database file created but is empty"
        except Exception as e:
            return f"Error reading file: {str(e)}"
    
    def search(self, query: Dict[str, Any]) -> Dict[str, Any]:
        """
        Search the database using OpenAI
        
        Args:
            query: Dictionary containing search parameters
        
        Returns:
            Dictionary with search results
        """
        if not self.client:
            return {"error": "OpenAI client not initialized"}
        
        try:
            prompt = self._build_prompt(query)
            
            response = self.client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": self.role_description},
                    {"role": "user", "content": prompt}
                ],
                temperature=0.3,
                max_tokens=2000
            )
            
            result = response.choices[0].message.content
            
            return {
                "agent": self.name,
                "query": query,
                "results": result,
                "status": "success"
            }
            
        except Exception as e:
            return {
                "agent": self.name,
                "query": query,
                "error": str(e),
                "status": "error"
            }
    
    def _build_prompt(self, query: Dict[str, Any]) -> str:
        """Build the prompt for OpenAI"""
        prompt = f"""
        You are {self.name}. Search the following database for information related to this query:
        
        Query Parameters: {json.dumps(query, indent=2)}
        
        Database Content:
        {self.database_content[:5000]}
        
        Provide detailed, relevant information found in the database. If no relevant information is found, state that clearly.
        Format your response in a structured way.
        """
        return prompt
    
    def chat(self, message: str, conversation_history: List[Dict] = None, context: Dict = None) -> Dict[str, Any]:
        """
        Chat with the agent - maintains conversation context
        
        Args:
            message: User's message
            conversation_history: Previous messages in the conversation
            context: Current context (extracted entities, etc.)
        
        Returns:
            Response with answer and updated context
        """
        if not self.client:
            return {"error": "OpenAI client not initialized"}
        
        # Reload database to ensure fresh data
        self.database_content = self.load_database()
        
        if not self.database_content or "Error" in self.database_content:
            return {
                "status": "error",
                "response": f"Database is not accessible. Please check the {self.name} database file.",
                "context": context or {}
            }
        
        try:
            # Build conversation messages for OpenAI
            system_prompt = f"""{self.role_description}

You have access to this database with real data:
---DATABASE START---
{self.database_content[:4000]}
---DATABASE END---

Instructions:
1. Search the database content above to answer questions
2. Provide specific details from the database (names, dates, phone numbers, etc.)
3. If you find relevant records, quote them directly
4. Remember context from previous messages in this conversation
5. If information is not in the database, say so clearly
6. Always be helpful and specific with the data you find

Current context from conversation: {context if context else 'None'}
"""
            
            messages = [{"role": "system", "content": system_prompt}]
            
            # Add conversation history if provided
            if conversation_history:
                for msg in conversation_history[-5:]:  # Last 5 messages for context
                    messages.append({
                        "role": msg["role"],
                        "content": msg["content"]
                    })
            
            # Add current message
            messages.append({"role": "user", "content": message})
            
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0.3,
                max_tokens=1000
            )
            
            answer = response.choices[0].message.content
            
            # Extract entities from the conversation
            new_context = self._extract_context(message, answer, context or {})
            
            return {
                "status": "success",
                "response": answer,
                "context": new_context,
                "agent": self.name
            }
            
        except Exception as e:
            return {
                "status": "error",
                "error": str(e),
                "response": f"An error occurred: {str(e)}. Please try again."
            }
    
    def _extract_context(self, user_message: str, assistant_response: str, current_context: Dict) -> Dict:
        """Extract and update context from conversation"""
        import re
        
        # Simple entity extraction (you can make this more sophisticated)
        context = current_context.copy()
        
        # Extract phone numbers
        phone_pattern = r'\+?\d{1,3}[-.\s]?\(?\d{1,4}\)?[-.\s]?\d{1,4}[-.\s]?\d{1,4}'
        phones = re.findall(phone_pattern, user_message + " " + assistant_response)
        if phones:
            context["phones"] = list(set(context.get("phones", []) + phones))
        
        # Extract names (simple pattern - words starting with capital letters)
        name_pattern = r'\b[A-Z][a-z]+ [A-Z][a-z]+\b'
        names = re.findall(name_pattern, user_message + " " + assistant_response)
        if names:
            context["names"] = list(set(context.get("names", []) + names))
        
        # Extract dates
        date_pattern = r'\d{4}-\d{2}-\d{2}'
        dates = re.findall(date_pattern, user_message + " " + assistant_response)
        if dates:
            context["dates"] = list(set(context.get("dates", []) + dates))
        
        return context

chat_manager.py - Manages chat sessions and conversations
from typing import Dict, List, Any
import json
from datetime import datetime

class ChatManager:
    def __init__(self):
        """Initialize chat manager with conversation storage"""
        self.conversations = {}  # Store conversations by session_id
        
    def create_session(self, agent_type: str) -> str:
        """Create a new chat session"""
        session_id = f"{agent_type}_{datetime.now().timestamp()}"
        self.conversations[session_id] = {
            "agent_type": agent_type,
            "messages": [],
            "context": {},
            "created_at": datetime.now().isoformat()
        }
        return session_id
    
    def add_message(self, session_id: str, role: str, content: str, context: Dict = None):
        """Add a message to the conversation"""
        if session_id not in self.conversations:
            return False
        
        message = {
            "role": role,  # "user" or "assistant"
            "content": content,
            "timestamp": datetime.now().isoformat()
        }
        
        if context:
            message["context"] = context
            # Update session context
            self.conversations[session_id]["context"].update(context)
        
        self.conversations[session_id]["messages"].append(message)
        return True
    
    def get_conversation(self, session_id: str) -> Dict:
        """Get full conversation history"""
        return self.conversations.get(session_id, None)
    
    def get_context(self, session_id: str) -> Dict:
        """Get current context for a session"""
        if session_id in self.conversations:
            return self.conversations[session_id]["context"]
        return {}
    
    def get_recent_messages(self, session_id: str, count: int = 5) -> List[Dict]:
        """Get recent messages from conversation"""
        if session_id in self.conversations:
            return self.conversations[session_id]["messages"][-count:]
        return []





leo_manager.py - Manages LEO (intelligence cards) operations

import json
import os
from datetime import datetime
from typing import Dict, List, Any, Optional
from uuid import uuid4

class LEOManager:
    def __init__(self, orchestrator):
        self.orchestrator = orchestrator
        self.leos_file = "data/leos.json"
        self.leos = self.load_leos()
        
    def load_leos(self) -> Dict:
        """Load saved LEOs from file"""
        if os.path.exists(self.leos_file):
            try:
                with open(self.leos_file, 'r') as f:
                    return json.load(f)
            except:
                return {}
        return {}
    
    def save_leos(self):
        """Save LEOs to file"""
        os.makedirs("data", exist_ok=True)
        with open(self.leos_file, 'w') as f:
            json.dump(self.leos, f, indent=2, default=str)
    
    def create_leo(self, 
                   name: str,
                   description: str,
                   target_identifiers: Dict,
                   selected_agents: List[str],
                   schedule: Optional[str] = None) -> str:
        """Create a new LEO"""
        leo_id = f"leo_{uuid4().hex[:8]}"
        
        self.leos[leo_id] = {
            "id": leo_id,
            "name": name,
            "description": description,
            "target_identifiers": target_identifiers,
            "selected_agents": selected_agents,
            "schedule": schedule or "manual",
            "created_at": datetime.now().isoformat(),
            "last_run": None,
            "status": "active",
            "results": []
        }
        
        self.save_leos()
        return leo_id
    
    def run_leo(self, leo_id: str) -> Dict:
        """Run a specific LEO"""
        if leo_id not in self.leos:
            return {"error": "LEO not found"}
        
        leo = self.leos[leo_id]
        results = {
            "leo_id": leo_id,
            "leo_name": leo["name"],
            "timestamp": datetime.now().isoformat(),
            "target": leo["target_identifiers"],
            "agent_results": {},
            "summary": ""
        }
        
        # Run only selected agents
        for agent_type in leo["selected_agents"]:
            if agent_type in self.orchestrator.agents:
                agent = self.orchestrator.agents[agent_type]
                
                # Reload text file for agent
                agent.database_content = agent.load_database()
                
                # Search with target identifiers
                agent_result = agent.search(leo["target_identifiers"])
                results["agent_results"][agent_type] = agent_result.get("results", "No data")
        
        # Generate summary
        results["summary"] = f"LEO '{leo['name']}' executed successfully. Queried {len(leo['selected_agents'])} agents."
        
        # Update last run time and save results
        self.leos[leo_id]["last_run"] = datetime.now().isoformat()
        self.leos[leo_id]["last_results"] = results  # Save for viewing
        self.save_leos()
        
        return results
    
    def get_all_leos(self) -> List[Dict]:
        """Get all LEOs"""
        return list(self.leos.values())
    
    def delete_leo(self, leo_id: str) -> bool:
        """Delete a LEO"""
        if leo_id in self.leos:
            del self.leos[leo_id]
            self.save_leos()
            return True
        return False

unified_agent.py - Unified search across all databases

from typing import Dict, Any, List
from .base_agent import BaseAgent
import json

class UnifiedAgent(BaseAgent):
    def __init__(self, orchestrator):
        """
        Initialize unified agent that can access all other agents
        
        Args:
            orchestrator: Reference to the orchestrator with all agents
        """
        role_description = """
        You are a Unified Intelligence Agent with access to multiple databases:
        1. Travel Database - travel records, co-travelers, destinations
        2. Telecom Database - call records, SMS, communication patterns  
        3. Tech Data Database - emails, purchases, digital footprint
        4. Meta Data Database - IP addresses, locations, device info
        5. Blog Database - public information, reputation, social media
        
        When asked for comprehensive information, you should search ALL relevant databases
        and provide a complete intelligence report.
        """
        
        # Don't need a specific db_path since we use all agents
        super().__init__("Unified Agent", "unified", role_description)
        self.orchestrator = orchestrator
        
    def chat(self, message: str, conversation_history: List[Dict] = None, context: Dict = None) -> Dict[str, Any]:
        """
        Chat with unified agent - queries all relevant agents
        """
        if not self.client:
            return {
                "status": "error",
                "response": "OpenAI client not initialized",
                "context": context or {}
            }
        
        # Determine which agents to query based on the message
        agents_to_query = self._determine_agents(message)
        
        # Collect data from all relevant agents
        all_data = {}
        for agent_type in agents_to_query:
            agent = self.orchestrator.agents.get(agent_type)
            if agent:
                # Get the agent's text file content
                agent_data = agent.load_database()
                if agent_data and "Error" not in agent_data:
                    all_data[agent_type] = agent_data
                    print(f"[Unified] Loaded {len(agent_data)} chars from {agent_type}")
        
        if not all_data:
            return {
                "status": "error",
                "response": "Unable to access any agent databases",
                "context": context or {}
            }
        
        try:
            # Build comprehensive prompt with all data
            system_prompt = f"""{self.role_description}
            
You have access to the following databases:

"""
            for agent_type, data in all_data.items():
                system_prompt += f"""
=== {agent_type.upper()} DATABASE ===
{data[:2000]}  # Limit each to prevent token overflow
===END {agent_type.upper()}===

"""
            
            system_prompt += """
Instructions:
1. Search through ALL the databases above for relevant information
2. When finding information, specify which database it came from
3. Provide comprehensive results from all relevant databases
4. Organize the response by category (Travel, Telecom, Tech, Meta, Blog)
5. If some databases don't have relevant info, mention that briefly
"""
            
            messages = [{"role": "system", "content": system_prompt}]
            
            # Add conversation history
            if conversation_history:
                for msg in conversation_history[-3:]:  # Less history due to larger context
                    messages.append({
                        "role": msg["role"],
                        "content": msg["content"]
                    })
            
            messages.append({"role": "user", "content": message})
            
            print(f"[Unified] Querying with data from {len(all_data)} agents")
            
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=0.3,
                max_tokens=1500  # More tokens for comprehensive response
            )
            
            answer = response.choices[0].message.content
            new_context = self._extract_context(message, answer, context or {})
            
            return {
                "status": "success",
                "response": answer,
                "context": new_context,
                "agent": self.name,
                "sources": list(all_data.keys())
            }
            
        except Exception as e:
            return {
                "status": "error",
                "response": f"Error processing unified query: {str(e)}",
                "context": context or {}
            }
    
    def _determine_agents(self, message: str) -> List[str]:
        """Determine which agents to query based on message content"""
        message_lower = message.lower()
        
        # Keywords for each agent type
        keywords = {
            'travel': ['travel', 'trip', 'flight', 'destination', 'co-traveler'],
            'telecom': ['call', 'phone', 'sms', 'communication', 'contact'],
            'tech': ['email', 'purchase', 'digital', 'whatsapp', 'telegram'],
            'meta': ['ip', 'location', 'device', 'metadata'],
            'blog': ['blog', 'reputation', 'public', 'social', 'news']
        }
        
        # If asking for "all" or "everything", query all agents
        if any(word in message_lower for word in ['all', 'everything', 'complete', 'full', 'comprehensive']):
            return ['travel', 'telecom', 'tech', 'meta', 'blog']
        
        # Otherwise, determine based on keywords
        agents = []
        for agent_type, words in keywords.items():
            if any(word in message_lower for word in words):
                agents.append(agent_type)
        
        # Default to all if no specific keywords found
        if not agents:
            agents = ['travel', 'telecom', 'tech', 'meta', 'blog']
        
        return agents
