    // Panel switching
    function switchPanel(panelName) {
        document.querySelectorAll('.content-panel').forEach(p => p.classList.remove('active'));
        document.getElementById(`${panelName}-panel`).classList.add('active');
        document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active'));
        // Highlight the matching nav item (safe even when called programmatically)
        const navItem = event && event.target && event.target.closest('.nav-item');
        if (navItem) {
            navItem.classList.add('active');
        } else {
            // Called programmatically — find nav item by panel name
            document.querySelectorAll('.nav-item').forEach(i => {
                if (i.getAttribute('onclick') && i.getAttribute('onclick').includes(`'${panelName}'`))
                    i.classList.add('active');
            });
        }
        currentPanel = panelName;
        if (panelName === 'targets') loadTargets();
        if (panelName === 'targeting') refreshTargetingWorkflows();
    }

    async function switchToAgentChat(agentType) {
        document.querySelectorAll('.content-panel').forEach(p => p.classList.remove('active'));
        const panel = document.getElementById(`${agentType}-chat-panel`);
        if (panel) panel.classList.add('active');
        document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active'));
        if (event && event.target) event.target.closest('.nav-item').classList.add('active');
        if (!currentSessions[agentType]) await startChatSession(agentType);
    }

    async function startChatSession(agentType) {
        try {
            const res = await fetch(`${API_BASE_URL}/chat/start?agent_type=${agentType}`, { method: 'POST' });
            const data = await res.json();
            if (data.status === 'success') { currentSessions[agentType] = data.session_id; return data.session_id; }
        } catch (e) { console.error('Failed to start session:', e); }
        return null;
    }

    async function sendChatMessage(agentType) {
        const input = document.getElementById(`${agentType}-input`);
        if (!input) return;
        const message = input.value.trim();
        if (!message) return;
        if (!currentSessions[agentType]) { const sid = await startChatSession(agentType); if (!sid) return; }
        const sessionId = currentSessions[agentType];
        addMessageToChat(agentType, message, 'user');
        input.value = '';
        showTypingIndicator(agentType);
        try {
            const res = await fetch(`${API_BASE_URL}/chat/message?session_id=${sessionId}&message=${encodeURIComponent(message)}`, { method: 'POST' });
            const data = await res.json();
            hideTypingIndicator(agentType);
            if (data.status === 'success') {
                addMessageToChat(agentType, data.response, 'assistant');
                updateContext(agentType, data.context);
                if (agentType === 'unified') addActivity('Unified search completed');
            } else {
                addMessageToChat(agentType, 'Error: ' + (data.error || 'Unknown'), 'assistant');
            }
        } catch (e) {
            hideTypingIndicator(agentType);
            addMessageToChat(agentType, 'Failed to send. Check server.', 'assistant');
        }
    }

    function addMessageToChat(agentType, text, sender) {
        const div = document.getElementById(`${agentType}-messages`);
        if (!div) return;
        const empty = div.querySelector('.empty-state');
        if (empty) empty.remove();
        let formatted = text;
        if (sender === 'assistant') {
            formatted = text.replace(/### (.*?)(\n|$)/g, '<h4 style="color:var(--accent-color);margin:8px 0;">$1</h4>')
                .replace(/## (.*?)(\n|$)/g, '<h3 style="color:var(--accent-color);margin:8px 0;">$1</h3>')
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>').replace(/\n/g, '<br>');
        }
        const msg = document.createElement('div');
        msg.className = `message ${sender}`;
        msg.innerHTML = `<div style="white-space:pre-wrap;word-wrap:break-word;">${formatted}</div><div class="message-time">${new Date().toLocaleTimeString()}</div>`;
        div.appendChild(msg);
        div.scrollTop = div.scrollHeight;
    }

    function showTypingIndicator(t) {
        const d = document.getElementById(`${t}-messages`); if (!d) return;
        const i = document.createElement('div'); i.className = 'typing-indicator'; i.id = `${t}-typing`;
        i.innerHTML = '<div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div>';
        d.appendChild(i); d.scrollTop = d.scrollHeight;
    }
    function hideTypingIndicator(t) { const i = document.getElementById(`${t}-typing`); if (i) i.remove(); }

    function updateContext(agentType, context) {
        const d = document.getElementById(`${agentType}-context`); if (!d) return;
        if (!context || Object.keys(context).length === 0) { d.innerHTML = '<p style="color:rgba(255,255,255,0.5);font-size:11px;">No context yet</p>'; return; }
        let html = '';
        for (const [k, v] of Object.entries(context)) {
            if (Array.isArray(v)) v.forEach(item => { html += `<span class="context-item">${k}: ${item}</span>`; });
        }
        d.innerHTML = html || '<p style="color:rgba(255,255,255,0.5);font-size:11px;">No entities detected</p>';
    }

    function handleChatKeypress(e, t) { if (e.key === 'Enter') sendChatMessage(t); }
    function sendQuickMessage(t, m) { const i = document.getElementById(`${t}-input`); if (i) { i.value = m; sendChatMessage(t); } }

    async function runLEO(leoId) {
        // If this LEO is already running, just reopen the progress panel
        if (isPipelineRunning('leo', leoId)) { reopenPipelinePanel(); return; }
        const btn = event.target; const orig = btn.innerHTML; btn.disabled = true; btn.innerHTML = '⏳ Running...';
        try {
            const res = await fetch(`${API_BASE_URL}/leo/run/${leoId}?background=true`, { method: 'POST' });
            const data = await res.json();
            btn.disabled = false; btn.innerHTML = orig;
            if (data.status === 'started' && data.run_id) {
                addActivity(`LEO started: ${data.leo_name}`);
                showLiveEventModal(leoId, data.run_id);
                connectToSSE(data.run_id, leoId);
                refreshLEOs();
            } else if (data.status === 'success' && data.results) {
                latestLEOResults = data.results;
                showResultsModal(data.results);
                refreshLEOs();
                addActivity(`LEO executed: ${data.results.leo_name}`);
            }
            else alert('Error: ' + (data.detail || 'Unknown'));
        } catch (e) { btn.disabled = false; btn.innerHTML = orig; alert('Failed: ' + e.message); }
    }

    function closePipelinePanel() {
        const panel = document.getElementById('pipelineFloat');

    function reopenPipelinePanel() {
        if (!pipelineRunContext) return;
        document.getElementById('pipelineMini').classList.remove('active');
        const panel = document.getElementById('pipelineFloat');
        // Restore title & status
        document.getElementById('pfTitle').textContent = (pipelineRunContext.type === 'targeting' ? '🎯 ' : '🎯 ') + pipelineRunContext.name;
        if (pipelineRunContext.done) {
            document.getElementById('pfStatus').textContent = pipelineRunContext.isError ? 'Error' : 'Complete';
            document.getElementById('pfStatus').className = 'pf-status ' + (pipelineRunContext.isError ? 'error' : 'done');
            document.getElementById('pfActions').style.display = pipelineRunContext.isError ? 'none' : 'flex';
        } else {
            document.getElementById('pfStatus').textContent = 'Running';
            document.getElementById('pfStatus').className = 'pf-status';
            document.getElementById('pfActions').style.display = 'none';
        }
        // Replay buffered events into the log
        const log = document.getElementById('pfEventLog');
        log.innerHTML = '';
        for (const ev of pipelineRunContext.events) {
            renderEventToLog(log, ev);
        }
        panel.classList.remove('minimized');
        panel.classList.add('active');
    }

    function renderEventToLog(log, event) {
        const t = new Date(event.timestamp * 1000).toLocaleTimeString();
        const icon = EVENT_ICONS[event.event_type] || '📌';
        const entry = document.createElement('div');
        entry.className = `event-entry ${event.event_type}`;
        entry.innerHTML = `<span class="event-time">${t}</span><span class="event-icon">${icon}</span><span class="event-msg">${event.message || event.event_type}</span>`;
        log.appendChild(entry);
        log.scrollTop = log.scrollHeight;
    }

    function appendFloatEvent(event) {
        // Buffer every event for replay
        if (pipelineRunContext) pipelineRunContext.events.push(event);
        // Render to visible log if panel is open
        const log = document.getElementById('pfEventLog');
        if (!log) return;
        if (log.querySelector('div[style]')) log.innerHTML = '';
        renderEventToLog(log, event);
        // Update mini indicator text with latest message
        if (pipelineRunContext && !pipelineRunContext.done) {
            document.getElementById('pipelineMiniText').textContent = event.message || (pipelineRunContext.name + ' running...');
        }
    }

    function onPipelineComplete(isError) {
        if (pipelineRunContext) { pipelineRunContext.done = true; pipelineRunContext.isError = isError; }
        const status = document.getElementById('pfStatus');
        const actions = document.getElementById('pfActions');
        status.textContent = isError ? 'Error' : 'Complete';
        status.className = 'pf-status ' + (isError ? 'error' : 'done');
        if (!isError) actions.style.display = 'flex';
        // Hide mini indicator and update it
        const mini = document.getElementById('pipelineMini');
        if (mini.classList.contains('active')) {
            document.getElementById('pipelineMiniText').textContent = (pipelineRunContext?.name || 'Pipeline') + (isError ? ' — Error' : ' — Complete!');
            mini.style.animation = 'none';
            mini.style.background = isError ? 'linear-gradient(135deg, #f44336, #d32f2f)' : 'linear-gradient(135deg, #4CAF50, #388E3C)';
        }
        // Refresh card grids to remove running indicators
        refreshTargetingWorkflows();
        refreshLEOs();
    }

    function isPipelineRunning(type, id) {

    }

    function connectToSSE(runId, leoId) {
        const evtSource = new EventSource(`${API_BASE_URL}/events/${runId}`);
        evtSource.onmessage = (e) => {
            try { appendFloatEvent(JSON.parse(e.data)); appendEventToLog(runId, JSON.parse(e.data)); } catch(err) {}
        };
        for (const etype of Object.keys(EVENT_ICONS)) {
            evtSource.addEventListener(etype, (e) => {
                try {
                    const event = JSON.parse(e.data);
                    appendFloatEvent(event);
                    appendEventToLog(runId, event);
                    if (etype === 'pipeline_complete' || etype === 'pipeline_error') {
                        evtSource.close();
                        onPipelineComplete(etype === 'pipeline_error');
                        if (leoId) {
                            setTimeout(async () => {
                                try {
                                    const res = await fetch(`${API_BASE_URL}/leo/${leoId}/results`);
                                    const data = await res.json();
                                    if (data.last_results) {
                                        latestLEOResults = data.last_results;
                                        data.last_results._run_history = data.run_history || [];
                                    }
                                } catch(fetchErr) {}
                                refreshLEOs();
                            }, 500);
                        }
                    }
                } catch(err) {}
            });
        }
        evtSource.onerror = () => { evtSource.close(); };
    }

    function appendEventToLog(runId, event) {
        const log = document.getElementById(`eventLog-${runId}`);
        if (!log) return;
        if (log.querySelector('div[style]')) log.innerHTML = '';

        const icon = EVENT_ICONS[event.event_type] || '📌';
        const entry = document.createElement('div');
        entry.className = `event-entry ${event.event_type}`;
        entry.innerHTML = `<span class="event-time">${t}</span><span class="event-icon">${icon}</span><span class="event-msg">${event.message || event.event_type}</span>`;
        log.appendChild(entry);
        log.scrollTop = log.scrollHeight;
    }

    async function runTargetingWorkflow(wfId) {
        // If this workflow is already running, just reopen the progress panel
        if (isPipelineRunning('targeting', wfId)) { reopenPipelinePanel(); return; }
        const btn = event.target; const orig = btn.innerHTML; btn.disabled = true; btn.innerHTML = '⏳ Running...';
        try {
            const res = await fetch(`${API_BASE_URL}/targeting/run/${wfId}?background=true`, { method: 'POST' });
            const data = await res.json();
            btn.disabled = false; btn.innerHTML = orig;
            if (data.status === 'started' && data.run_id) {
                addActivity(`Targeting started: ${data.workflow_name}`);
                showLiveTargetingModal(wfId, data.run_id);
                connectToTargetingSSE(data.run_id, wfId);
                refreshTargetingWorkflows();  // Refresh cards to show running indicator
            } else if (data.status === 'success' && data.results) {
                showTargetingResultsModal(data.results);
                refreshTargetingWorkflows();
                addActivity(`Targeting complete: ${data.results.workflow_name}`);
            } else alert('Error: ' + (data.detail || 'Unknown'));
        } catch (e) { btn.disabled = false; btn.innerHTML = orig; alert('Failed: ' + e.message); }
    }

    function connectToTargetingSSE(runId, wfId) {
        const evtSource = new EventSource(`${API_BASE_URL}/events/${runId}`);
        evtSource.onmessage = (e) => {
            try { appendFloatEvent(JSON.parse(e.data)); appendEventToLog(runId, JSON.parse(e.data)); } catch(err) {}
        };
        for (const etype of Object.keys(EVENT_ICONS)) {
            evtSource.addEventListener(etype, (e) => {
                try {
                    const event = JSON.parse(e.data);
                    appendFloatEvent(event);
                    appendEventToLog(runId, event);
                    if (etype === 'pipeline_complete' || etype === 'pipeline_error') {
                        evtSource.close();
                        onPipelineComplete(etype === 'pipeline_error');
                        if (wfId) {
                            setTimeout(async () => {
                                try {
                                    const res = await fetch(`${API_BASE_URL}/targeting/${wfId}/results`);
                                    const data = await res.json();
                                    if (data.last_results) {
                                        latestTargetingResults = data.last_results;
                                        data.last_results._run_history = data.run_history || [];
                                    }
                                } catch(fetchErr) {}
                                refreshTargetingWorkflows();
                            }, 500);
                        }
                    }
                } catch(err) {}
            });
        }
        evtSource.onerror = () => { evtSource.close(); };
    }

                    <div style="display:none;padding:12px;background:rgba(0,0,0,0.2);border-radius:0 0 8px 8px;margin-top:-4px;"><div style="white-space:pre-wrap;line-height:1.5;font-size:12px;max-height:300px;overflow-y:auto;">${formatReportText(content)}</div></div>
                </div>`;
            }
            html += `<div class="agent-result-section"><h3 style="color:var(--accent-color);margin-bottom:10px;">📂 5. RAW DATA (Per-Agent) — Click to expand</h3>${rawHtml}</div>`;
        }

        // Pipeline events
        if (results.run_id) {
            html += `<div class="agent-result-section"><h3 style="color:var(--accent-color);margin-bottom:10px;">📡 Pipeline Events</h3><div id="eventLog-${results.run_id}" class="event-log"><div style="opacity:0.5;">Loading event history...</div></div></div>`;
            loadEventHistory(results.run_id);
        }

        html += `<div style="margin-top:15px;text-align:center;"><button onclick="exportTargetingResults()" style="padding:8px 20px;background:rgba(255,152,0,0.2);border:1px solid #FF9800;border-radius:5px;cursor:pointer;color:white;">📥 Export Report</button></div>`;
        document.getElementById('modalBody').innerHTML = html;
        document.getElementById('resultsModal').classList.add('active');
    }

    function formatReportText(text) {
        if (!text) return '';
        return text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
            .replace(/\*\*(.*?)\*\*/g,'<strong style="color:#B0E0E6;">$1</strong>')
            .replace(/\n/g,'<br>');
    }

    let latestTargetingResults = null;
    function exportTargetingResults() {
        const results = latestTargetingResults;
        if (!results) return;
        const sections = results.sections || {};
        let txt = `TARGETING REPORT EXPORT\n${'='.repeat(60)}\nWorkflow: ${results.workflow_name}\nTime: ${new Date(results.timestamp).toLocaleString()}\n`;
        if (results.target) { txt += `\nTarget:\n`; for (const [k,v] of Object.entries(results.target)) txt += `  ${k}: ${v}\n`; }
        if (sections.summary) txt += `\n1. SUMMARY\n${'-'.repeat(40)}\n${sections.summary}\n`;
        if (sections.access_summary) txt += `\n2.1 ACCESS\n${'-'.repeat(40)}\n${sections.access_summary}\n`;
        if (sections.accessibility_summary) txt += `\n2.2 ACCESSIBILITY\n${'-'.repeat(40)}\n${sections.accessibility_summary}\n`;
        if (sections.motivations_summary) txt += `\n2.3 MOTIVATIONS\n${'-'.repeat(40)}\n${sections.motivations_summary}\n`;
        if (sections.suitability_summary) txt += `\n2.4 SUITABILITY\n${'-'.repeat(40)}\n${sections.suitability_summary}\n`;
        if (sections.pattern_of_life) txt += `\n3. PATTERN OF LIFE\n${'-'.repeat(40)}\n${sections.pattern_of_life}\n`;
        if (sections.security_ci) txt += `\n4. SECURITY & CI\n${'-'.repeat(40)}\n${sections.security_ci}\n`;
        if (results.final_report) txt += `\n${'='.repeat(60)}\nFULL REPORT:\n${'='.repeat(60)}\n${results.final_report}\n`;
        const blob = new Blob([txt], { type: 'text/plain' });
        const a = document.createElement('a'); a.href = URL.createObjectURL(blob);
        a.download = `targeting_report_${Date.now()}.txt`; a.click();
    }
