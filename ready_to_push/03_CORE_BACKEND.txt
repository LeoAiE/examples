"""
Multi-LLM Provider Abstraction Layer
Supports OpenAI, Anthropic (Claude), Google (Gemini), and local LLMs.
Provides unified interface with automatic failover.
"""

import time
from typing import Dict, Any, List, Optional
from dataclasses import dataclass, field


@dataclass
class LLMMetrics:
    """Track per-request metrics."""
    provider: str = ""
    model: str = ""
    latency_ms: float = 0
    input_tokens: int = 0
    output_tokens: int = 0
    cost_estimate: float = 0.0
    success: bool = True
    error: str = ""


@dataclass
class LLMConfig:
    """Configuration for an LLM provider."""
    provider: str  # openai, anthropic, google, local
    api_key: str = ""
    model: str = ""
    base_url: str = ""  # For local LLMs (OpenAI-compatible endpoint)
    temperature: float = 0.3
    max_tokens: int = 2000


class LLMProvider:
    """Unified LLM interface with failover support."""

    def __init__(self, configs: List[LLMConfig], timeout: int = 30):
        self.configs = {c.provider: c for c in configs}
        self.timeout = timeout
        self.metrics_log: List[LLMMetrics] = []
        self._clients: Dict[str, Any] = {}
        self._init_clients()

    def _init_clients(self):
        """Initialize available LLM clients."""
        for provider, cfg in self.configs.items():
            try:
                if provider == "openai" and cfg.api_key:
                    from openai import OpenAI
                    kwargs = {"api_key": cfg.api_key}
                    if cfg.base_url:
                        kwargs["base_url"] = cfg.base_url
                    self._clients["openai"] = OpenAI(**kwargs)
                elif provider == "anthropic" and cfg.api_key:
                    import anthropic
                    self._clients["anthropic"] = anthropic.Anthropic(api_key=cfg.api_key)
                elif provider == "google" and cfg.api_key:
                    import google.generativeai as genai
                    genai.configure(api_key=cfg.api_key)
                    self._clients["google"] = genai
                elif provider == "local" and cfg.base_url:
                    from openai import OpenAI
                    self._clients["local"] = OpenAI(
                        api_key="local", base_url=cfg.base_url
                    )
            except Exception:
                pass  # Provider not available

    def chat(
        self,
        messages: List[Dict[str, str]],
        provider: Optional[str] = None,
        model: Optional[str] = None,
        temperature: Optional[float] = None,
        max_tokens: Optional[int] = None,
        stream: bool = False,
    ) -> Dict[str, Any]:
        """
        Send a chat completion request. Falls back through providers on failure.

        Args:
            messages: List of {"role": ..., "content": ...} dicts
            provider: Specific provider to use (None = try default then failover)
            model: Override model name
            temperature: Override temperature
            max_tokens: Override max tokens
            stream: Whether to stream the response

        Returns:
            {"content": str, "metrics": LLMMetrics}
        """
        providers_to_try = (
            [provider] if provider and provider in self._clients
            else list(self._clients.keys())
        )

        last_error = ""
        for prov in providers_to_try:
            cfg = self.configs.get(prov)
            if not cfg:
                continue

            m = model or cfg.model
            t = temperature if temperature is not None else cfg.temperature
            mt = max_tokens or cfg.max_tokens
            metrics = LLMMetrics(provider=prov, model=m)
            start = time.time()

            try:
                if prov in ("openai", "local"):
                    result = self._chat_openai(prov, messages, m, t, mt)
                elif prov == "anthropic":
                    result = self._chat_anthropic(messages, m, t, mt)
                elif prov == "google":
                    result = self._chat_google(messages, m, t, mt)
                else:
                    continue

                metrics.latency_ms = (time.time() - start) * 1000
                metrics.input_tokens = result.get("input_tokens", 0)
                metrics.output_tokens = result.get("output_tokens", 0)
                self.metrics_log.append(metrics)

                return {"content": result["content"], "metrics": metrics}

            except Exception as e:
                metrics.success = False
                metrics.error = str(e)
                metrics.latency_ms = (time.time() - start) * 1000
                self.metrics_log.append(metrics)
                last_error = str(e)
                continue

        return {
            "content": f"All LLM providers failed. Last error: {last_error}",
            "metrics": LLMMetrics(success=False, error=last_error),
        }

    def _chat_openai(self, provider_key, messages, model, temperature, max_tokens):
        client = self._clients[provider_key]
        response = client.chat.completions.create(
            model=model,
            messages=messages,
            temperature=temperature,
            max_tokens=max_tokens,
        )
        choice = response.choices[0]
        usage = response.usage
        return {
            "content": choice.message.content,
            "input_tokens": usage.prompt_tokens if usage else 0,
            "output_tokens": usage.completion_tokens if usage else 0,
        }

    def _chat_anthropic(self, messages, model, temperature, max_tokens):
        client = self._clients["anthropic"]
        # Anthropic uses system as a separate param
        system_msg = ""
        chat_msgs = []
        for msg in messages:
            if msg["role"] == "system":
                system_msg += msg["content"] + "\n"
            else:
                chat_msgs.append(msg)

        kwargs = {
            "model": model or "claude-sonnet-4-20250514",
            "max_tokens": max_tokens,
            "temperature": temperature,
            "messages": chat_msgs,
        }
        if system_msg:
            kwargs["system"] = system_msg.strip()

        response = client.messages.create(**kwargs)
        return {
            "content": response.content[0].text,
            "input_tokens": response.usage.input_tokens,
            "output_tokens": response.usage.output_tokens,
        }

    def _chat_google(self, messages, model, temperature, max_tokens):
        genai = self._clients["google"]
        gmodel = genai.GenerativeModel(model or "gemini-pro")

        # Convert messages to Gemini format
        history = []
        last_content = ""
        for msg in messages:
            if msg["role"] == "system":
                history.append({"role": "user", "parts": [msg["content"]]})
                history.append({"role": "model", "parts": ["Understood."]})
            elif msg["role"] == "user":
                last_content = msg["content"]
            elif msg["role"] == "assistant":
                history.append({"role": "model", "parts": [msg["content"]]})

        chat = gmodel.start_chat(history=history)
        response = chat.send_message(
            last_content,
            generation_config={"temperature": temperature, "max_output_tokens": max_tokens},
        )
        return {
            "content": response.text,
            "input_tokens": 0,
            "output_tokens": 0,
        }

    def get_metrics_summary(self) -> Dict[str, Any]:
        """Get summary of all LLM usage metrics."""
        if not self.metrics_log:
            return {"total_requests": 0}

        total = len(self.metrics_log)
        successes = sum(1 for m in self.metrics_log if m.success)
        avg_latency = sum(m.latency_ms for m in self.metrics_log) / total
        total_input = sum(m.input_tokens for m in self.metrics_log)
        total_output = sum(m.output_tokens for m in self.metrics_log)

        by_provider = {}
        for m in self.metrics_log:
            if m.provider not in by_provider:
                by_provider[m.provider] = {"requests": 0, "successes": 0}
            by_provider[m.provider]["requests"] += 1
            if m.success:
                by_provider[m.provider]["successes"] += 1

        return {
            "total_requests": total,
            "successes": successes,
            "failures": total - successes,
            "avg_latency_ms": round(avg_latency, 2),
            "total_input_tokens": total_input,
            "total_output_tokens": total_output,
            "by_provider": by_provider,
        }

# --- Target Card ---
# Background pipeline results storage
_pipeline_results: Dict[str, Any] = {}


@app.post("/target-card")
async def create_target_card(request: TargetCardRequest,
                              background: bool = False):
    try:
        card_kwargs = {}
        for field in ["first_name", "first_name_variations", "father_name",
                       "father_name_variations", "third_name", "third_name_variations",
                       "fourth_name", "fourth_name_variations", "native_name",
                       "kunya", "aliases", "phone", "email", "passport", "ip_address", "name"]:
            val = getattr(request, field, None)
            if val:
                card_kwargs[field] = val

        if background:
            # Generate run_id upfront so client can subscribe to SSE
            from tools.shared.target_store import TargetStore
            identifiers = {k: v for k, v in card_kwargs.items() if v}
            target_id = TargetStore.generate_target_id(identifiers)
            run_id = f"run_pending_{id(request)}"

            def _run_pipeline():
                try:
                    results = orchestrator.create_target_card(
                        event_bus=event_bus, **card_kwargs)
                    actual_run_id = results.get("run_id", run_id)
                    _pipeline_results[actual_run_id] = results
                except Exception as exc:
                    from tools.shared.event_bus import PipelineEvent
                    event_bus.emit(PipelineEvent(
                        event_type="pipeline_error",
                        run_id=run_id,
                        message=str(exc),
                    ))

            thread = threading.Thread(target=_run_pipeline, daemon=True)
            thread.start()
            return {"status": "started", "target_id": target_id,
                    "message": "Pipeline running in background. Connect to /events/{run_id} for live updates."}

        # Synchronous mode (default)
        results = orchestrator.create_target_card(event_bus=event_bus, **card_kwargs)
        run_id = results.get("run_id", "")
        if run_id:
            _pipeline_results[run_id] = results
        return {"status": "success", "results": results}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/agentic-search")
async def run_agentic_search(request: AgenticSearchRequest):
    """Run an autonomous agent search using the tool-calling engine.
    The AI agent decides which tools to call (cross-agent queries, etc.)
    in a recursive loop until it has a complete analysis.

    Best for: security, counter_intel, access, accessibility, motivation, suitability agents.
    These agents have cross-agent access and can query each other's databases."""
    if request.agent not in orchestrator.agents:
        raise HTTPException(status_code=404, detail=f"Agent '{request.agent}' not found")
    try:
        agent = orchestrator.agents[request.agent]
        result = agent.agentic_search(request.query,
                                       identifiers=request.identifiers or {})
        return {"status": "success", "results": result}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

    if background:
        # Non-blocking: start in background thread, return run_id immediately
        import asyncio
        from concurrent.futures import ThreadPoolExecutor
        leo = leo_manager.leos[leo_id]
        target_ids = leo["target_identifiers"]
        target_store = orchestrator.target_store
        target_id, _ = target_store.get_or_create_target(target_ids)
        run_id = target_store.start_run(target_id, leo["selected_agents"], leo_id=leo_id)

        def _run_in_bg():
            try:
                leo_manager.run_leo(leo_id, event_bus=event_bus,
                                    run_id=run_id, target_id=target_id)
            except Exception as e:
                print(f"[LEO BG] Error running {leo_id}: {e}")

        loop = asyncio.get_event_loop()
        loop.run_in_executor(None, _run_in_bg)

        return {
            "status": "started",
            "run_id": run_id,
            "leo_id": leo_id,
            "leo_name": leo["name"],
            "message": "LEO execution started in background. Subscribe to SSE for live updates.",
        }
    else:
        # Blocking: wait for completion (original behavior)
        result = leo_manager.run_leo(leo_id, event_bus=event_bus)
        if "error" in result:
            raise HTTPException(status_code=404, detail=result["error"])
        return {"status": "success", "results": result}

@app.get("/events/{run_id}")
async def stream_events(run_id: str):
    """SSE endpoint for live pipeline events.

    Frontend connects via EventSource(GET /events/{run_id}).
    Replays history for late-connecting clients, then streams live.
    """
    async def event_generator():
        queue = await event_bus.subscribe(run_id)
        try:
            while True:
                try:
                    event = await asyncio.wait_for(queue.get(), timeout=30.0)
                    yield event.to_sse()
                    # Stop streaming after pipeline completes or errors
                    if event.event_type in ("pipeline_complete", "pipeline_error"):
                        break
                except asyncio.TimeoutError:
                    # Send keepalive comment to prevent connection drop
                    yield ": keepalive\n\n"
        finally:
            event_bus.unsubscribe(run_id, queue)

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )

    success = target_store.delete_record(record_id, reason=reason)
    if success:
        return {"status": "deleted", "record_id": record_id,
                "message": "Record tombstoned — will not be re-imported"}
    raise HTTPException(status_code=500, detail="Delete failed")


@app.post("/targets/{target_id}/records/{record_id}/restore")
async def restore_record(target_id: str, record_id: int):
    """Restore a deleted record and remove tombstone."""
    target_store = orchestrator.target_store
    record = target_store.get_record_detail(record_id)
    if not record or record.get("target_id") != target_id:
        raise HTTPException(status_code=404, detail="Record not found")
    success = target_store.restore_record(record_id)
    if success:
        return {"status": "restored", "record_id": record_id}
    raise HTTPException(status_code=500, detail="Restore failed")
