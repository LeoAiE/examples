#!/usr/bin/env python3
"""
AI Agent Platform - Air-Gap Setup Script
=========================================
Run this FIRST on the target machine to create the entire folder structure.
Then copy the text files and follow SETUP_INSTRUCTIONS.txt to populate.

Usage:
    python 00_SETUP_SCRIPT.py
    (or)
    python 00_SETUP_SCRIPT.py /custom/install/path
"""

import os
import sys

# Default install location (current directory)
BASE = sys.argv[1] if len(sys.argv) > 1 else os.path.join(os.getcwd(), "ai-agent-platform")

# =========================================================================
# FOLDER STRUCTURE
# =========================================================================
DIRECTORIES = [
    # Root
    "V2",

    # Backend root
    "V2/backend",

    # Agents - all 16 agent files + orchestrator + managers
    "V2/backend/agents",

    # Data - all database .txt files, .csv, .json, .db
    "V2/backend/data",

    # Tools root
    "V2/backend/tools",

    # Shared tools - used by ALL agents (name generator, voter search, etc.)
    "V2/backend/tools/shared",

    # Agent-specific tools - tools only used by one agent
    "V2/backend/tools/agent_specific",
    "V2/backend/tools/agent_specific/travel",
    "V2/backend/tools/agent_specific/sigint",
    "V2/backend/tools/agent_specific/fisa",
    "V2/backend/tools/agent_specific/sna",

    # Database connectors - text file and JSON readers
    "V2/backend/tools/database_connectors",

    # API integrations - OSINT, reverse lookups (placeholder stubs)
    "V2/backend/tools/api_integrations",

    # Frontend
    "V2/frontend",
]

# =========================================================================
# __init__.py FILES (empty files needed for Python imports)
# =========================================================================
INIT_FILES = [
    "V2/backend/agents/__init__.py",
    "V2/backend/tools/__init__.py",
    "V2/backend/tools/shared/__init__.py",
    "V2/backend/tools/agent_specific/travel/__init__.py",
    "V2/backend/tools/agent_specific/sigint/__init__.py",
    "V2/backend/tools/agent_specific/fisa/__init__.py",
    "V2/backend/tools/agent_specific/sna/__init__.py",
    "V2/backend/tools/database_connectors/__init__.py",
    "V2/backend/tools/api_integrations/__init__.py",
]

# =========================================================================
# .env TEMPLATE
# =========================================================================
ENV_TEMPLATE = """# =========================================================================
# AI Agent Platform - Environment Configuration
# =========================================================================

# --- OPTION A: Air-Gapped / Single LLM Endpoint ---
# If these are set, the platform uses ONE endpoint for everything.
# Works with any OpenAI-compatible API (vLLM, Ollama, text-generation-inference, etc.)
# LLM_BASE_URL=https://your-internal-llm.local/v1
# LLM_API_KEY=your-key-here
# LLM_MODEL=your-model-name
# LLM_PROVIDER=openai

# --- OPTION B: Cloud Providers (multi-provider with failover) ---
OPENAI_API_KEY=sk-your-openai-key-here
ANTHROPIC_API_KEY=sk-ant-your-anthropic-key-here
# GOOGLE_API_KEY=your-google-key-here

# Default provider: openai | anthropic | google
DEFAULT_LLM_PROVIDER=openai
DEFAULT_MODEL=gpt-4o-mini

# --- Engine Settings ---
# Max LLM calls per target (safety limit)
LLM_MAX_CALLS_PER_TARGET=300

# Tool-calling engine iterations before stopping
TOOL_CALLING_MAX_ITERATIONS=10
TOOL_CALLING_MAX_MISTAKES=3

# Sub-agent processing
SUB_AGENT_RECORDS_PER_WORKER=2
SUB_AGENT_MAX_WORKERS=50
"""

# =========================================================================
# requirements.txt
# =========================================================================
REQUIREMENTS = """openai>=1.51.0
anthropic>=0.39.0
google-generativeai>=0.8.0
fastapi>=0.104.0
uvicorn>=0.24.0
python-dotenv>=1.0.0
pydantic>=2.5.0
"""

# =========================================================================
# main.py
# =========================================================================
MAIN_PY = '''"""Entry point - run with: python main.py"""
import uvicorn

if __name__ == "__main__":
    uvicorn.run("api_server:app", host="0.0.0.0", port=8000, reload=True)
'''

# =========================================================================
# RUN IT
# =========================================================================
def main():
    print(f"\\n{'='*60}")
    print(f"  AI Agent Platform - Setup Script")
    print(f"  Installing to: {BASE}")
    print(f"{'='*60}\\n")

    # Create directories
    for d in DIRECTORIES:
        path = os.path.join(BASE, d)
        os.makedirs(path, exist_ok=True)
        print(f"  [DIR]  {d}/")

    # Create __init__.py files
    for f in INIT_FILES:
        path = os.path.join(BASE, f)
        if not os.path.exists(path):
            with open(path, "w") as fh:
                fh.write("")
            print(f"  [INIT] {f}")

    # Create .env template
    env_path = os.path.join(BASE, "V2/backend/.env")
    if not os.path.exists(env_path):
        with open(env_path, "w") as f:
            f.write(ENV_TEMPLATE)
        print(f"  [ENV]  V2/backend/.env (EDIT THIS with your API keys!)")

    # Create requirements.txt
    req_path = os.path.join(BASE, "V2/backend/requirements.txt")
    with open(req_path, "w") as f:
        f.write(REQUIREMENTS)
    print(f"  [REQ]  V2/backend/requirements.txt")

    # Create main.py
    main_path = os.path.join(BASE, "V2/backend/main.py")
    with open(main_path, "w") as f:
        f.write(MAIN_PY)
    print(f"  [MAIN] V2/backend/main.py")

    print(f"\\n{'='*60}")
    print(f"  DONE! Folder structure created.")
    print(f"")
    print(f"  NEXT STEPS:")
    print(f"  1. Edit V2/backend/.env with your API keys")
    print(f"  2. Copy code from the text files (see SETUP_INSTRUCTIONS.txt)")
    print(f"  3. cd {os.path.join(BASE, 'V2/backend')}")
    print(f"  4. pip install -r requirements.txt")
    print(f"  5. python main.py")
    print(f"{'='*60}\\n")


if __name__ == "__main__":
    main()
