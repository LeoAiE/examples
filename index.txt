    // Start chat session
    async function startChatSession(agentType) {
        try {
            const response = await fetch(`${API_BASE_URL}/chat/start?agent_type=${agentType}`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                }
            });
            
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            
            const data = await response.json();
            
            if (data.status === 'success') {
                currentSessions[agentType] = data.session_id;
                console.log(`Started session for ${agentType}: ${data.session_id}`);
                return data.session_id;
            } else {
                console.error('Failed to start session:', data);
                alert(`Failed to start chat session for ${agentType}`);
                return null;
            }
        } catch (error) {
            console.error('Failed to start chat session:', error);
            alert(`Error starting chat session: ${error.message}`);
            return null;
        }
    }
    
    // Send chat message
    async function sendChatMessage(agentType) {
        const input = document.getElementById(`${agentType}-input`);
        if (!input) {
            console.error(`Input not found for ${agentType}`);
            return;
        }
        
        const message = input.value.trim();
        if (!message) return;
        
        // Ensure we have a session
        if (!currentSessions[agentType]) {
            console.log('No session found, creating one...');
            const sessionId = await startChatSession(agentType);
            if (!sessionId) {
                alert('Failed to start chat session. Please try again.');
                return;
            }
        }
        
        const sessionId = currentSessions[agentType];
        console.log(`Sending message to ${agentType} with session ${sessionId}: ${message}`);
        
        // Add user message to chat
        addMessageToChat(agentType, message, 'user');
        
        // Clear input
        input.value = '';
        
        // Show typing indicator
        showTypingIndicator(agentType);
        
        try {
            const response = await fetch(`${API_BASE_URL}/chat/message?session_id=${sessionId}&message=${encodeURIComponent(message)}`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                }
            });
            
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            
            const data = await response.json();
            
            // Remove typing indicator
            hideTypingIndicator(agentType);
            
            if (data.status === 'success') {
                // Add assistant response
                addMessageToChat(agentType, data.response, 'assistant');
                
                // Update context
                updateContext(agentType, data.context);
                
                // Add to activity feed if this is the unified agent
                if (agentType === 'unified') {
                    addActivity(`Unified search completed`);
                }
            } else {
                addMessageToChat(agentType, 'Sorry, an error occurred: ' + (data.error || 'Unknown error'), 'assistant');
            }
        } catch (error) {
            console.error('Error sending message:', error);
            hideTypingIndicator(agentType);
            addMessageToChat(agentType, 'Failed to send message. Please check the server.', 'assistant');
        }
    }
    
    // Add message to chat
    function addMessageToChat(agentType, text, sender) {
        const messagesDiv = document.getElementById(`${agentType}-messages`);
        if (!messagesDiv) {
            console.error(`Messages div not found for ${agentType}`);
            return;
        }
        
        // Remove empty state if exists
        const emptyState = messagesDiv.querySelector('.empty-state');
        if (emptyState) {
            emptyState.remove();
        }
        
        // Format the text - convert markdown to HTML
        let formattedText = text;
        if (sender === 'assistant') {
            // Convert markdown to HTML
            formattedText = text
                // Headers
                .replace(/### (.*?)(\n|$)/g, '<h4 style="color: #87CEEB; margin: 10px 0;">$1</h4>')
                .replace(/## (.*?)(\n|$)/g, '<h3 style="color: #87CEEB; margin: 10px 0;">$1</h3>')
                .replace(/# (.*?)(\n|$)/g, '<h2 style="color: #87CEEB; margin: 10px 0;">$1</h2>')
                // Bold text
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                // Line breaks
                .replace(/\n/g, '<br>')
                // Lists (basic)
                .replace(/^\d+\. /gm, '<br>• ')
                .replace(/^- /gm, '<br>• ');
        }
        
        const messageDiv = document.createElement('div');
        messageDiv.className = `message ${sender}`;
        messageDiv.innerHTML = `
            <div style="white-space: pre-wrap; word-wrap: break-word;">
                ${formattedText}
            </div>
            <div class="message-time">${new Date().toLocaleTimeString()}</div>
        `;
        
        messagesDiv.appendChild(messageDiv);
        messagesDiv.scrollTop = messagesDiv.scrollHeight;
    }

    // Update context display
    function updateContext(agentType, context) {
        const contextDiv = document.getElementById(`${agentType}-context`);
        if (!contextDiv) return;
        
        if (!context || Object.keys(context).length === 0) {
            contextDiv.innerHTML = '<p style="color: rgba(255,255,255,0.5); font-size: 12px;">No context yet</p>';
            return;
        }
        
        let html = '';
        for (const [key, value] of Object.entries(context)) {
            if (Array.isArray(value)) {
                value.forEach(item => {
                    html += `<span class="context-item">${key}: ${item}</span>`;
                });
            }
        }
        contextDiv.innerHTML = html || '<p style="color: rgba(255,255,255,0.5); font-size: 12px;">No entities detected</p>';
    }

    async function runLEO(leoId) {
        try {
            // Find the button that was clicked
            const button = event.target;
            const originalText = button.innerHTML;
            button.disabled = true;
            button.innerHTML = '⏳ Running...';
            
            const response = await fetch(`${API_BASE_URL}/leo/run/${leoId}`, {
                method: 'POST'
            });
            
            const data = await response.json();
            
            // Restore button
            button.disabled = false;
            button.innerHTML = originalText;
            
            if (data.status === 'success' && data.results) {
                // Store results
                latestLEOResults = data.results;
                
                // Show results modal
                showResultsModal(data.results);
                
                // Refresh LEO grid to update last run time
                refreshLEOs();
                
                // Add to activity feed
                addActivity(`LEO executed: ${data.results.leo_name || 'Unknown LEO'}`);
            } else {
                alert('Error running LEO: ' + (data.detail || 'Unknown error'));
            }
        } catch (error) {
            alert('Failed to run LEO: ' + error.message);
        }
    }

    async function deleteLEO(leoId) {
        if (!confirm('Are you sure you want to delete this LEO?')) return;
        
        try {
            const response = await fetch(`${API_BASE_URL}/leo/delete/${leoId}`, {
                method: 'DELETE'
            });
            
            if (response.ok) {
                alert('LEO deleted successfully');
                refreshLEOs();
            }
        } catch (error) {
            alert('Failed to delete LEO: ' + error.message);
        }
    }

                }
            });

        // Load LEOs on page load
        refreshLEOs();
        
        // Initialize dashboard stats
        updateDashboardStats();
        setInterval(updateDashboardStats, 60000); // Update every minute
        
        // Reset search counter at midnight
        const now = new Date();
        const tomorrow = new Date(now);
        tomorrow.setDate(now.getDate() + 1);
        tomorrow.setHours(0, 0, 0, 0);
        const msUntilMidnight = tomorrow.getTime() - now.getTime();
        
        setTimeout(() => {
            localStorage.setItem('searchesToday', '0');
            updateDashboardStats();

            setInterval(() => {

    function exportResults() {
        if (!latestLEOResults) return;
        
        // Create a text version of results
        let textContent = `LEO Results Export\n`;
        textContent += `==================\n\n`;
        textContent += `LEO Name: ${latestLEOResults.leo_name}\n`;
        textContent += `Executed: ${new Date(latestLEOResults.timestamp).toLocaleString()}\n\n`;
        
        textContent += `Target Information:\n`;
        for (const [key, value] of Object.entries(latestLEOResults.target || {})) {
            textContent += `  ${key}: ${value}\n`;
        }
        textContent += `\n`;
        
        for (const [agentType, results] of Object.entries(latestLEOResults.agent_results || {})) {
            textContent += `\n${agentType.toUpperCase()} AGENT RESULTS:\n`;
            textContent += `${'='.repeat(50)}\n`;
            textContent += results;
            textContent += `\n\n`;
        }
        
        // Create download
        const blob = new Blob([textContent], { type: 'text/plain' });
        const url = window.URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `leo_results_${Date.now()}.txt`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        window.URL.revokeObjectURL(url);
    }

    window.addEventListener('click', (event) => {
        const modal = document.getElementById('resultsModal');
        if (event.target === modal) {
            closeResultsModal();
        }
    });
</script>
